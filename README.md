import time

def run_flora_care():
    print("🌱 أهلاً بك في نظام المُزارع الذكي (FloraCare-AI) لرعاية النباتات 🌱")
    print("-----------------------------------------------------------------")

    # 1. إدخال البيانات (محاكاة قراءة الحساسات الذكية)
    plant_name = input("🌿 أدخل اسم النبتة (مثال: نعناع، جوري، صبار): ").strip()
    soil_type = input("⏳ نوع التربة (طينية / رملية / بيتموس): ").strip()
    
    try:
        temp = float(input("🌡️ أدخل درجة حرارة الجو الحالية (°م): "))
        soil_moisture = float(input("💧 أدخل نسبة رطوبة التربة الحالية (%): "))
    except ValueError:
        print("❌ خطأ: يرجى إدخال أرقام صحيحة لدرجة الحرارة والرطوبة.")
        return

    print("\n🤖 يقوم الذكاء الاصطناعي الآن بتحليل البيانات وبيئة النبات...")
    time.sleep(1.5)

    # 2. خوارزمية اتخاذ القرار الذكي لحماية النبات وتوفير مياه الري
    if soil_moisture < 30:
        status = "🚨 جفاف شديد خطر على الجذور!"
        if temp > 35:
            advice = f"💧 ارْوِ نبتة {plant_name} فوراً بكمية مياه وفيرة، وانقلها إلى الظل لأن الحرارة مرتفعة جداً ({temp}°م)."
        else:
            advice = f"💧 نبتة {plant_name} بحاجة للري المعتدل الآن."
    elif 30 <= soil_moisture <= 70:
        status = "✅ رطوبة مثالية ومستقرة."
        advice = f"👍 وضع نبتة {plant_name} ممتاز! لا تقم بريها اليوم لتوفير المياه والمحافظة على سلامة الجذور."
    else:
        status = "⚠️ تشبع زائد بالماء (خطر تعفن الجذور)."
        advice = f"🚫 أوقف الري تماماً عن نبتة {plant_name}، وتأكد من وجود فتحات لتصريف المياه الزائدة أسفل حوض الزراعة."

    # 3. عرض النتائج والتقرير البيئي للجنة
    print("-----------------------------------------------------------------")
    print(f"📊 التقرير الذكي لنبتة ({plant_name}):")
    print(f"⬅️ حالة رطوبة التربة: {status}")
    print(f"⬅️ نوع التربة المستخدمة: {soil_type}")
    print(f"📢 الإجراء المطلوب فوراً: {advice}")
    print("-----------------------------------------------------------------")

if __name__ == "__main__":
    run_flora_care()
